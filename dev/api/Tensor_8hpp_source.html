502 struct exclude_first;

504 template <typename First, typename... Rest>
505 struct exclude_first<std::variant<First, Rest...>> {
506 using type = std::variant<Rest...>;

513 make_variant_from_transform_t<typename internal::exclude_first<Type_list>::type,
514 std::add_pointer>;

516 // convert this->_impl->_storage._impl->Mem to a typed variant of pointers, excluding void*
517 pointer_types ptr() const;
518
519 // Convert storage to the logical Cytnx element pointer type.
520 //
521 // Use ptr_as<T>() for host code and raw memory copies. For complex tensors, T is
522 // std::complex<...>. Use gpu_ptr_as<T>() for CUDA kernel code; for complex tensors, T is
523 // cuda::std::complex<...>. CUDA library APIs that require cuComplex ABI pointers should cast
524 // explicitly at the library-call boundary.
525 template <typename T>
526 T *ptr_as() const {
527 cytnx_error_msg(this->dtype() != Type_class::cy_typeid_v<std::remove_cv_t<T>>,
528 "[ERROR] Attempt to convert dtype %d (%s) to pointer of type %s",
529 this->dtype(), Type_class::getname(this->dtype()).c_str(),
530 Type_class::getname(Type_class::cy_typeid_v<std::remove_cv_t<T>>).c_str());
531 return static_cast<T *>(this->_impl->_storage._impl->data());
532 }
533
534 #ifdef UNI_GPU
535 // std::variant of pointers to Type_list_gpu, without void ....
536 using gpu_pointer_types =
537 make_variant_from_transform_t<typename internal::exclude_first<Type_list_gpu>::type,
538 std::add_pointer>;
539
540 // convert this->_impl->_storage->Mem to a typed variant of pointers, excluding void*
541 gpu_pointer_types gpu_ptr() const;
542
543 // Convert storage to the CUDA kernel element pointer type.
544 //
545 // This is for CUDA kernel interfaces. Generic host code should use ptr_as<T>() so the logical
546 // Cytnx dtype mapping is checked. CUDA library APIs that require cuComplex ABI pointers should
547 // cast explicitly at the library-call boundary.
548 template <typename T>
549 T *gpu_ptr_as() const {
550 cytnx_error_msg(
551 this->dtype() != Type_class::cy_typeid_gpu_v<std::remove_cv_t<T>>,
552 "[ERROR] Attempt to convert dtype %d (%s) to GPU pointer of type %s", this->dtype(),
553 Type_class::getname(this->dtype()).c_str(),
554 Type_class::getname(Type_class::cy_typeid_gpu_v<std::remove_cv_t<T>>).c_str());
555 return static_cast<T *>(this->_impl->_storage._impl->data());
556 }

1506 void append(const Tensor &rhs) {
1507 // Tensor in;
1508 // check Tensor in shape:
1509 cytnx_error_msg(rhs.is_void() || this->is_void(), "[ERROR] try to append a null Tensor.%s",
1510 "\n");
1511 cytnx_error_msg(this->is_scalar(), "[ERROR] try to append to a rank-0 Tensor.%s", "\n");
1512 if (!this->is_contiguous()) this->contiguous_();
1513
1514 cytnx_error_msg(rhs.rank() != this->rank() - 1,
1515 "[ERROR] try to append a Tensor with rank not match.%s", "\n");
1516 cytnx_uint64 Nelem = 1;
1517 for (unsigned int i = 0; i < rhs.shape().size(); i++) {
1518 cytnx_error_msg(rhs.shape()[i] != this->shape()[i + 1],
1519 "[ERROR] dimension mismatch @ rhs.rank: [%d] this: [%d] rhs: [%d]\n", i,
1520 this->shape()[i + 1], rhs.shape()[i]);
1521 Nelem *= rhs.shape()[i];
1522 }
1523
1524 // check type:
1525 Tensor in;
1526 if (rhs.dtype() != this->dtype()) {
1527 in = rhs.astype(this->dtype());
1528 if (!in.is_contiguous()) in.contiguous_();
1529 } else {
1530 if (!rhs.is_contiguous())
1531 in = rhs.contiguous();
1532 else
1533 in = rhs;
1534 }
1535 this->_impl->_shape[0] += 1;
1536 cytnx_uint64 oldsize = this->_impl->_storage.size();
1537 this->_impl->_storage.resize(oldsize + Nelem);
1538 memcpy(((char *)this->_impl->_storage.data()) +
1539 oldsize * Type.typeSize(this->dtype()) / sizeof(char),
1540 in._impl->_storage.data(), Type.typeSize(in.dtype()) * Nelem);
1541 }

1786 // Scalar-like operand concepts for the free Tensor / UniTensor arithmetic and comparison
1787 // operators. They live here, after the class, because cytnx_scalar_like names Tensor::Tproxy
1788 // and so needs Tensor to be complete; linalg.hpp includes this header and reuses them.
1789 //
1790 // Each concept admits exactly the operand types the corresponding operators are instantiated
1791 // for, so an unsupported operand is rejected at overload resolution instead of surviving to a
1792 // link error. Widening one without adding the matching explicit instantiation reintroduces
1793 // precisely that failure (#1003, operator hygiene -- Ian's review).
1794
1796 // Plain scalar values: the cytnx dtype scalars and cytnx::Scalar. This is the operand set of
1797 // `operator%` (Tensor and UniTensor) and `operator==` (Tensor), which have no element-proxy
1798 // instantiations -- see Mod.cpp and Cpr.cpp.
1799 template <class T>
1800 concept cytnx_scalar_value =
1801 CytnxType<std::remove_cvref_t<T>> || std::is_same_v<std::remove_cvref_t<T>, Scalar>;
1802
1803 // Adds both element proxies: the operand set of the free Tensor `+ - * /` operators.
1804 template <class T>
1805 concept cytnx_scalar_like =
1806 cytnx_scalar_value<T> || std::is_same_v<std::remove_cvref_t<T>, Tensor::Tproxy> ||
1807 std::is_same_v<std::remove_cvref_t<T>, Scalar::Sproxy>;
1808
1809 // Adds only the Scalar proxy: the operand set of the free UniTensor `+ - * /` operators, which
1810 // are instantiated for Scalar::Sproxy but never for Tensor::Tproxy -- see Add.cpp.
1811 template <class T>
1812 concept cytnx_unitensor_scalar_like =
1813 cytnx_scalar_value<T> || std::is_same_v<std::remove_cvref_t<T>, Scalar::Sproxy>;
